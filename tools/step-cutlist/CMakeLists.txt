cmake_minimum_required(VERSION 3.20)
project(StepCutlist LANGUAGES CXX)
find_package(OpenCASCADE REQUIRED)
add_executable(step-cutlist main.cpp)
target_compile_features(step-cutlist PRIVATE cxx_std_17)
target_include_directories(step-cutlist SYSTEM PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_link_libraries(step-cutlist PRIVATE TKernel TKMath TKBRep TKTopAlgo TKPrim TKBO TKXSBase TKDESTEP TKXCAF TKLCAF TKCDF)
target_compile_options(step-cutlist PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(cutlist-tests tests.cpp)
  target_compile_features(cutlist-tests PRIVATE cxx_std_17)
  target_include_directories(cutlist-tests SYSTEM PRIVATE ${OpenCASCADE_INCLUDE_DIR})
  target_link_libraries(cutlist-tests PRIVATE TKernel TKMath TKBRep TKTopAlgo TKPrim TKBO TKXSBase TKDESTEP TKXCAF TKLCAF TKCDF)
  add_test(NAME geometry_and_optimizer COMMAND cutlist-tests ${CMAKE_CURRENT_BINARY_DIR}/fixture.step)
  set_tests_properties(geometry_and_optimizer PROPERTIES FIXTURES_SETUP cutlist_fixture)
  add_test(NAME inventory_and_layout COMMAND ${CMAKE_COMMAND} -DTOOL=$<TARGET_FILE:step-cutlist> -DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/fixture-plan -DINPUT=${CMAKE_CURRENT_BINARY_DIR}/fixture.step -P ${CMAKE_CURRENT_SOURCE_DIR}/run-integration.cmake)
  set_tests_properties(inventory_and_layout PROPERTIES FIXTURES_REQUIRED cutlist_fixture)
endif()

if(APPLE)
  target_link_libraries(step-cutlist PRIVATE "-framework CoreText" "-framework CoreFoundation" "-framework CoreGraphics")
  if(BUILD_TESTING)
    target_link_libraries(cutlist-tests PRIVATE "-framework CoreText" "-framework CoreFoundation" "-framework CoreGraphics")
  endif()
endif()

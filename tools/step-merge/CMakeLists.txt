cmake_minimum_required(VERSION 3.20)
project(StepMerge LANGUAGES CXX)
find_package(OpenCASCADE REQUIRED)
add_executable(step-merge step_merge.cpp)
target_compile_features(step-merge PRIVATE cxx_std_17)
target_include_directories(step-merge SYSTEM PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_link_libraries(step-merge PRIVATE TKernel TKMath TKBRep TKTopAlgo TKXSBase TKDESTEP TKXCAF TKLCAF TKCDF)
add_executable(check-orientation check_orientation.cpp)
target_compile_features(check-orientation PRIVATE cxx_std_17)
target_include_directories(check-orientation SYSTEM PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_link_libraries(check-orientation PRIVATE TKernel TKMath TKBRep TKTopAlgo TKXSBase TKDESTEP TKXCAF TKLCAF TKCDF)
